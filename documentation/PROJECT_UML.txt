' --- COMBAT AND ENTITIES ---

abstract class Entity {
    # name: string
    # maxHP: int
    # currentHP: int
    # level: int
    # critRate: float
    # critMultiplier: float
    + {static} totalEntities: int
    + getName(): string
    + getHP(): int
    + getMaxHP(): int
    + getLevel(): int
    + takeDamage(amount: int): void
    + isDead(): bool
    + calculateDamage(base: int): int
    + {abstract} takeTurn(): void
}

class Player {
    - xp: int
    - gold: int
    - mana: int
    - inventory: Inventory*
    - spellbook: Spellbook*
    - activeModifiers: StatModifier*
    - numModifiers: int
    - modifierCapacity: int
    - equippedWeapon: Weapon*
    - equippedArmor: Armor*
    + attack(target: Entity*): void
    + useItem(index: int): void
    + castSpell(index: int, target: Entity*): void
    + equip(item: Equippable*): void
    + applyModifier(mod: StatModifier): void
    + clearModifiers(): void
    + takeTurn(): void
}

abstract class Enemy {
    - lootTable: LootTable*
    - activeEffects: StatusEffect*
    - numEffects: int
    - effectCapacity: int
    + applyEffect(effect: StatusEffect): void
    + {abstract} decideAction(): int
    + {abstract} takeTurn(): void
    + dropLoot(): LootBundle
}

class Goblin {
    - stealthChance: float
    + decideAction(): int
    + takeTurn(): void
}

class Knight {
    - blockChance: float
    + decideAction(): int
    + takeTurn(): void
}

class Boss {
    - phase: int
    - chargeCounter: int
    + triggerSpecial(): void
    + decideAction(): int
    + takeTurn(): void
}

class BattleManager <<Singleton>> {
    - {static} instance: BattleManager*
    - player: Player*
    - enemy: Enemy*
    - active: bool
    - BattleManager()
    + {static} getInstance(): BattleManager*
    + startBattle(p: Player*, e: Enemy*): void
    + processTurn(): void
    + calculateFinalDamage(attacker: Entity*, target: Entity*): int
}

class CombatRules {
    + {static} maxRounds: int
    + {static} baseCritBonus: float
    + {static} dodgeChance: float
}

Entity <|-- Player
Entity <|-- Enemy
Enemy <|-- Goblin
Enemy <|-- Knight
Enemy <|-- Boss

BattleManager --> Player
BattleManager --> Enemy
BattleManager ..> CombatRules


' --- ITEM SYSTEM ---

abstract class Item {
    # name: string
    # value: int
    + {static} totalItemsCreated: int
    + getName(): string
    + getValue(): int
}

abstract class Consumable {
    + {abstract} use(target: Player*): void
}

abstract class Equippable {
    # durability: int
    # maxDurability: int
    + {abstract} onEquip(target: Player*): void
    + onHit(): void
}

class Weapon {
    - baseDamage: int
    + onEquip(target: Player*): void
    + onHit(): void
}

class Armor {
    - baseDefense: int
    + onEquip(target: Player*): void
    + onHit(): void
}

class Material {
    - type: MaterialType
}

class HealthPotion {
    - healAmount: int
    + use(target: Player*): void
}

class ManaPotion {
    - restoreAmount: int
    + use(target: Player*): void
}

class DamagePotion {
    - damageBuffAmount: int
    + use(target: Player*): void
}

class DefensePotion {
    - defenseBuffAmount: int
    + use(target: Player*): void
}

class Inventory {
    - capacity: int
    - numItems: int
    - items: Item**
    + canFit(item: Item*): bool
    + addItem(item: Item*): bool
    + removeItem(index: int): void
}

class ItemFactory <<Singleton>> {
    - {static} instance: ItemFactory*
    - ItemFactory()
    + {static} getInstance(): ItemFactory*
    + createWeapon(): Weapon*
    + createArmor(): Armor*
    + createPotion(): Consumable*
    + createMaterial(): Material*
}

class ItemRules {
    + {static} maxStackSize: int
    + {static} defaultSellRatio: float
}

Item <|-- Consumable
Item <|-- Equippable
Item <|-- Material

Consumable <|-- HealthPotion

Consumable <|-- DamagePotion
Consumable <|-- DefensePotion

Player *-- Inventory
Player o-- Weapon : equips
Player o-- Armor : equips
Inventory o-- Item

ItemFactory ..> ItemRules


' --- SPELL SYSTEM ---

    # manaCost: int
    + {static} globalManaTax: float
    + cast(caster: Entity*, target: Entity*): void
    + {abstract} effect(caster: Entity*, target: Entity*): void

    - damage: int
    + cast(caster: Entity*, target: Entity*): void
    + effect(caster: Entity*, target: Entity*): void

    + cast(caster: Entity*, target: Entity*): void

class Spellbook {

' --- MAP AND ROOMS ---

class Room {
    - narrativeText: string
    - enemy: Enemy*
    - store: Store*
    + enter(player: Player*): void
    + interact(player: Player*): void
}

class TreasureRoom {
    - rewardGold: int
    + enter(player: Player*): void
    + interact(player: Player*): void
}

class MapGrid {
    - width: int
    - height: int
    - rooms: Room***
    - playerPosition: Coordinate
    + move(dx: int, dy: int): bool
    + render(): void
}

class Coordinate {
    + x: int
    + y: int
}

class WorldGenerator <<Singleton>> {
    - {static} instance: WorldGenerator*
    - WorldGenerator()
    + {static} getInstance(): WorldGenerator*
    + generateMap(w: int, h: int): MapGrid*
    + placeRooms(map: MapGrid*): void
}

class WorldRules {
    + {static} worldSeed: int
    + {static} mapWidth: int
    + {static} mapHeight: int
}

MapGrid *-- Room
Room <|-- TreasureRoom
MapGrid --> Coordinate
WorldGenerator ..> MapGrid
WorldGenerator ..> WorldRules
Room o-- Enemy
Room o-- Store


' -- ECONOMY AND LOOT --

class LootBundle {
    + xp: int
    + gold: int
    + items: Item**
    + numItems: int
}

class DropEntry {
    + item: Item*
    + chance: float
}

class LootTable {
    - entries: DropEntry*
    - numEntries: int
    - capacity: int
    + roll(): LootBundle
}

class Store {
    - inventory: Item**
    - numItems: int
    - capacity: int
    + buy(player: Player*, index: int): void
    + sell(player: Player*, index: int): void
}

Enemy --> LootTable
LootTable *-- DropEntry
DropEntry o-- Item
Store o-- Item
